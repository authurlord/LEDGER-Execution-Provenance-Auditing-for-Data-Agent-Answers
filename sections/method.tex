\section{The LEDGER Audit Layer}
\label{sec:method}

\etitle{Overview.}
Given a completed episode $E$, \ledger resolves its declared evidence,
reconciles observed reads, replays the final program in the resolved view,
and returns $\mathsf{Audit}(E)$ with a machine-readable witness or failure
record. When replay is unavailable, bounded closure over $\operands$ provides
a fallback. A task-specific interface independently produces $S(E;V_q)$;
the two statuses meet only at governance.

\subsection{Trace capture and evidence normalization}
\label{sec:tracecapture}

\etitle{Trace capture.}
The host wrapper freezes the final program, declared evidence, execution
status and output, observed data reads, exceptions, accessed attributes, and
operands reaching the returned result. Python/pandas and SQL adapters map
these records into the common trace $\tau$ defined in
Section~\ref{sec:episode}. Declared sources, host reads, and replay reads remain
separate fields.

\etitle{Evidence normalization.}
The resolver maps declarations to
$F_{\mathrm{res}}=R(F_{\mathrm{decl}})$ before comparing
$F_{\mathrm{host}}$ with the resolved view. This step implements evidence
authorization and supplies the data view used by restricted replay.

\subsection{Episode audit and restricted replay}
\label{sec:auditproc}

\etitle{Audit order.}
Algorithm~\ref{alg:audit} resolves evidence and checks host-read authorization
before replay, then records replay-boundary, execution, and output-comparison
conditions. A failure record $c$ may retain several observed conditions;
closure is considered only when replay is unavailable and no observed failure
already determines the outcome.

\begin{algorithm}[t]
\caption{\ledger episode audit}
\label{alg:audit}
\begin{algorithmic}[1]
\Require episode $E$, resolver $R$, replay runtime
$\mathcal R_{\mathrm{replay}}$, comparator $\operatorname{cmp}_{\tol}$,
operations $\opset$, depth $d$
\State $c\gets[\,]$; $b\gets\bot$
\State $F_{\mathrm{res}}\gets R(F_{\mathrm{decl}})$

\If{$F_{\mathrm{res}}$ is incomplete}
  \State append $\mathrm{unresolved\ source}$ to $c$
\Else
  \If{$F_{\mathrm{host}}\nsubseteq F_{\mathrm{res}}$}
    \State append $\mathrm{undeclared\ host\ read}$ to $c$
  \EndIf
\EndIf

\If{$s_{\mathrm{host}}\neq\mathrm{success}$}
  \State append $\mathrm{host\ execution}$ to $c$
\EndIf

\If{$F_{\mathrm{res}}$ is complete and $P^*$ is executable in
$\mathcal R_{\mathrm{replay}}$}
  \State $\rho\gets
  \operatorname{Exec}_{\mathcal R_{\mathrm{replay}}}
  (P^*,F_{\mathrm{res}})$

  \If{$F_{\mathrm{replay}}\nsubseteq F_{\mathrm{res}}$}
    \State append $\mathrm{replay\ boundary}$ to $c$
  \EndIf

  \If{$s_{\mathrm{replay}}\neq\mathrm{success}$}
    \State append $\mathrm{replay\ execution}$ to $c$
  \Else
    \State $b\gets
    \operatorname{cmp}_{\tol}(v_{\mathrm{replay}},\hat a)$

    \If{$b=0$}
      \State append $\mathrm{output\ mismatch}$ to $c$
    \ElsIf{$b=\bot$}
      \State append $\mathrm{missing\ comparable\ output}$ to $c$
    \EndIf
  \EndIf

  \If{$c=[\,]$ and $b=1$}
    \State \Return $\cert_{\mathrm{replay}}$ and replay witness
  \EndIf
\EndIf

\If{$c\neq[\,]$}
  \State \Return $\failure(c)$ and diagnostic record
\EndIf

\If{$P^*$ is not executable in $\mathcal R_{\mathrm{replay}}$
    and $\operands$ is complete}
  \State \Return
  \Call{BoundedClosure}{$\operands,\opset,d,\hat a$}
\EndIf

\State \Return $\unknown$
\end{algorithmic}
\end{algorithm}

\etitle{Replay witness.}
The replay runtime mounts only $F_{\mathrm{res}}$, records
$F_{\mathrm{replay}}$, and applies
$\operatorname{cmp}_{\tol}(v_{\mathrm{replay}},\hat a)$. Deterministic
normalization handles strings, booleans, collections, and structured outputs;
numeric values use tolerance $\tol$. A replay certificate stores $P^*$,
$F_{\mathrm{res}}$, host and replay reads, replay output, and comparator
configuration.

\etitle{Running example.}
For the identity-theft episode, all 17 declared files resolve, so
$|F_{\mathrm{decl}}|=|F_{\mathrm{res}}|=17$. The preserved final program
$P^*$ reads the national identity-theft table and 16 state-level
fraud-and-other tables. Because the archived episode did not retain syscall
reads, we reconstruct $F_{\mathrm{host}}$ from the exact program and verify it
with a fresh trace: $|F_{\mathrm{host}}|=17$ but
$|F_{\mathrm{host}}\cap F_{\mathrm{res}}|=1$, leaving 16 undeclared reads.

The program unions 87 metropolitan-area names from those 16 files, filters the
national table, infers population as
$\mathrm{pop}=\mathrm{reports}/\mathrm{rate}_{100\mathrm{k}}
\times100{,}000$, and retains 14 rows. Their report sum is 150,138, yielding
$v_{\mathrm{host}}=10{,}724.1429$ after rounding; this equals $\hat a$.

Restricted replay mounts only $F_{\mathrm{res}}$: the 16 fraud-file existence
checks fail, the valid-name set is empty, and the program exits normally with
\texttt{\{"main-task": null\}}. Thus
$s_{\mathrm{replay}}=\mathrm{success}$,
$F_{\mathrm{replay}}\subseteq F_{\mathrm{res}}$, and
$\operatorname{cmp}_{\tol}(v_{\mathrm{replay}},\hat a)=\bot$.
The failure record therefore contains both an undeclared-host-read witness and
a missing comparable replay output; no replay certificate is issued.

\subsection{Operand-constrained fallback}
\label{sec:closure}

\etitle{Fallback witness.}
When a runnable replay environment is unavailable, \ledger searches only over
the program-reached operands rather than all numbers in the cited context.
With
$\opset=\{\mathrm{sum},\mathrm{diff},\mathrm{ratio},
\mathrm{pctchg},\mathrm{prod}\}$,
the depth-$d$ closure recursively applies valid operations, normalizes results,
and removes duplicates.

\begin{definition}[Closure certificate]
\label{def:closure}
Episode $E$ receives a depth-$d$ closure certificate when
$\hat a\in\cl_d(\operands)$ within tolerance $\tol$. The witness stores a
minimal expression and its operand subset.
\end{definition}

\begin{algorithm}[t]
\caption{Bounded operand closure}
\label{alg:closure}
\begin{algorithmic}[1]
\Function{BoundedClosure}{$\operands,\opset,d,\hat a$}
\State $S_0\gets\operatorname{normalize}(\operands)$
\For{$j=1,\ldots,d$}
  \State $S_j\gets S_{j-1}$
  \ForAll{$x,y\in S_{j-1}$ and $o\in\opset$}
    \State add valid normalized $o(x,y)$ with an expression witness
  \EndFor
\EndFor
\If{$\hat a\in S_d$ within $\tol$}
  \State \Return $\cert_{\mathrm{closure}}$ and a minimal witness
\EndIf
\State \Return $\unknown$
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{proposition}[Closure completeness]
\label{prop:complete}
If $\hat a$ is generated from reached operands by operations in $\opset$,
expression depth at most $d$, and the audit's normalization and tolerance,
then $\hat a\in\cl_d(\operands)$.
\end{proposition}

\noindent\emph{Proof sketch.}
The operands form $S_0$. Induction on expression depth places every valid
internal result in the next closure level, so the root appears by depth $d$.
The full proof appears in~\ref{app:proofs}.

A closure certificate establishes derivability from $\operands$; it does not
carry the read-set guarantee of a replay certificate or identify the operation
requested by $q$.

\subsection{Semantic interface and governance}
\label{sec:semanticinterface}

\etitle{Semantic status.}
The semantic interface consumes $q$, $P^*$, and the task specification $V_q$
and returns $S(E;V_q)$. SQL tasks may compare normalized program structure or
execute programs on database variants \citep{zhong2020testsuite}; other tasks
may provide executable constraints, metamorphic relations, or human judgement.
This status remains separate from $\mathsf{Audit}(E)$.

\etitle{Governance.}
A policy maps the two statuses to deliver, abstain, or escalate. A provenance
certificate with incorrect semantics is withheld, while a failure or unknown
provenance status is abstained from or escalated. A certificate with unknown
semantic status is handled by deployment policy.

\subsection{Complexity and deterministic contract}

\etitle{Cost and determinism.}
Resolution and read reconciliation cost
$O(|F_{\mathrm{decl}}|+|F_{\mathrm{host}}|)$, and replay adds one execution
of $P^*$. If $n_j=|S_j|$ and $r=|\opset|$, closure level $j$ generates at
most $r n_{j-1}^{2}$ candidates before normalization and duplicate removal.
For a fixed trace, resolver, replay runtime, comparator, operation vocabulary,
and depth, \ledger is deterministic and invokes no language model.
