\section{Data-Agent Execution and Audit Setting}
\label{sec:problem}

\subsection{Data-agent environment and completed episode}
\label{sec:episode}

\etitle{Environment.}
A data-agent environment is
$\mathcal E=(\mathcal D,\mathcal K,\mathcal T,
\mathcal R_{\mathrm{host}})$, where $\mathcal D$ contains typed
executable data objects, $\mathcal K$ contains schemas, catalog metadata,
task instructions, and other planning knowledge, $\mathcal T$ is the tool
set, and $\mathcal R_{\mathrm{host}}$ executes data programs. Any object
from $\mathcal K$ read by the final program is registered in $\mathcal D$
and treated as an execution dependency.

\etitle{Agent output.}
A data agent $\mathcal A$ answers question $q$ through actions
$u_t=\mathcal A(q,h_{t-1})$, where $h_{t-1}$ contains earlier actions and
tool observations. On termination, it submits
$(P^*,F_{\mathrm{decl}},\hat a)$: the final executable program
$P^*\in\mathcal L$, declared evidence identifiers
$F_{\mathrm{decl}}\subseteq\mathcal I$, and stated answer
$\hat a\in\mathcal Y$. Our live-trace adapters support SQL and
Python/pandas; the bounded arithmetic language used without replay is
defined in Section~\ref{sec:closure}.

\etitle{Execution record.}
For runtime $\mathcal R$, program $P$, and mounted data view
$F\subseteq\mathcal D$, let
$\operatorname{Exec}_{\mathcal R}(P,F)
=(s,v,F_{\mathrm{read}},\operands)$ return execution status $s$, output
$v\in\mathcal Y\cup\{\bot\}$, observed reads $F_{\mathrm{read}}$, and the
multiset $\operands$ of scalar values reaching the final computation.
Here $\bot$ denotes the absence of a comparable output. The final host
execution is
$\operatorname{Exec}_{\mathcal R_{\mathrm{host}}}(P^*,\mathcal D)
=(s_{\mathrm{host}},v_{\mathrm{host}},F_{\mathrm{host}},\operands)$.
We freeze
$\tau=(P^*,F_{\mathrm{decl}},F_{\mathrm{host}},\operands,
 s_{\mathrm{host}},v_{\mathrm{host}})$ and define the completed episode as
$E=(q,\mathcal E,\tau,\hat a)$. The full log may additionally retain
program revisions, tool calls, exceptions, and timestamps.

\subsection{Audit task and trust model}
\label{sec:statuses}
\label{sec:threat}

\etitle{Resolved view and replay.}
The canonical resolver $R:\mathcal I\rightarrow\mathcal D\cup\{\bot\}$
maps each declared identifier to a registered data object. The resolved view
is $F_{\mathrm{res}}=\{R(f):f\in F_{\mathrm{decl}},R(f)\neq\bot\}$ and is
complete when every declaration resolves. Restricted replay produces
$\rho=\operatorname{Exec}_{\mathcal R_{\mathrm{replay}}}
(P^*,F_{\mathrm{res}})
=(s_{\mathrm{replay}},v_{\mathrm{replay}},F_{\mathrm{replay}},
\operands_{\mathrm{replay}})$.

\etitle{Audit outputs.}
A fixed comparator
$\operatorname{cmp}_{\tol}:\mathcal Y\times\mathcal Y
\rightarrow\{1,0,\bot\}$ returns agreement, mismatch, or non-comparability;
numeric outputs use tolerance $\tol$, and structured outputs use
deterministic normalization. The execution-provenance audit returns
$\mathsf{Audit}(E)\in
\{\cert_{\mathrm{replay}},\cert_{\mathrm{closure}},
\failure(c),\unknown\}$, where a certificate carries an execution or
expression witness, $c$ stores one or more observed failure codes and their
trace evidence, and $\unknown$ denotes that neither a certificate nor an
explicit failure is established.

Semantic validation independently returns
$S(E;V_q)\in\{\mathrm{correct},\mathrm{incorrect},\mathrm{unknown}\}$ for a
task-specific specification $V_q$, such as reference SQL, executable
constraints, database variants, metamorphic relations, or human judgement.
The provenance status evaluates the program--evidence--answer relation; the
semantic status evaluates the question--program relation.

% \etitle{Three audit relations.}
A completed episode induces evidence authorization
$F_{\mathrm{host}}\leftrightarrow F_{\mathrm{res}}$, execution
reproducibility $(P^*,F_{\mathrm{res}})\leftrightarrow\hat a$, and semantic
correctness $(q,V_q)\leftrightarrow P^*$. The first checks whether observed
host reads remain within the resolved declared view; the second checks whether
$P^*$ reproduces $\hat a$ in that view; the third checks whether $P^*$
implements the task specification. Section~\ref{sec:requirements} gives their
certificate conditions and guarantees.

\etitle{Trusted components.}
The host agent, generated program, declared evidence, host execution, and
stated answer are audited objects. \ledger trusts the trace logger, canonical
resolver, restricted replay runtime, and type-aware comparator. A certificate
is relative to the integrity and coverage of these components; tampered logs,
sandbox violations, external nondeterminism, or uninstrumented dependencies
produce $\unknown$ or a platform-level failure rather than a certificate.

\etitle{Running example.}
In the identity-theft episode, all 17 declarations resolve, so
$|F_{\mathrm{decl}}|=|F_{\mathrm{res}}|=17$. A reconstruction from the
preserved final program, confirmed by a fresh trace, gives
$|F_{\mathrm{host}}|=17$ but
$|F_{\mathrm{host}}\cap F_{\mathrm{res}}|=1$; hence
$F_{\mathrm{host}}\nsubseteq F_{\mathrm{res}}$. Host execution succeeds with
$v_{\mathrm{host}}=\hat a=10{,}724.1429$. Restricted replay also exits
normally, but returns \texttt{\{"main-task": null\}}, represented as
$v_{\mathrm{replay}}=\bot$ and
$\operatorname{cmp}_{\tol}(v_{\mathrm{replay}},\hat a)=\bot$. Thus
$\mathsf{Audit}(E)=\failure(c)$ with
$c=\{\mathrm{undeclared\ host\ read},
\mathrm{missing\ comparable\ output}\}$; no replay certificate is issued.
